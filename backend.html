Admin sign in

Sign in with the admin email registered in Firebase Authentication.

This page does not include a built-in username or password. Firebase Authentication and Firestore rules protect admin access.

Connect Firebase once

Replace the API-key placeholder in this file and in index.html. Then:

  1. Enable Email/Password in Firebase Authentication and create an admin email account.
  2. In Firestore, create users/{adminUid} with role: "admin".
  3. Create the database, publish the rules below, and host both pages on Firebase Hosting or another HTTPS host.
Starting Firestore security rules
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function admin() {
      return signedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }
    function owner(farmId) {
      return signedIn()
        && get(/databases/$(database)/documents/farms/$(farmId)).data.ownerUid == request.auth.uid;
    }
    function editor(farmId) {
      return signedIn()
        && ('editorEmails' in get(/databases/$(database)/documents/farms/$(farmId)).data)
        && request.auth.token.email in get(/databases/$(database)/documents/farms/$(farmId)).data.editorEmails;
    }
    match /users/{uid} {
      allow create: if signedIn() && request.auth.uid == uid
        && request.resource.data.role == 'farmer';
      allow read: if signedIn() && (request.auth.uid == uid || admin());
    }
    match /farms/{farmId} {
      allow read: if resource.data.isPublic == true || owner(farmId) || editor(farmId) || admin();
      allow create: if signedIn() && request.resource.data.ownerUid == request.auth.uid;
      allow update, delete: if owner(farmId) || editor(farmId) || admin();
      match /plants/{plantId} {
        allow read: if get(/databases/$(database)/documents/farms/$(farmId)).data.isPublic == true || owner(farmId) || editor(farmId) || admin();
        allow create, update, delete: if owner(farmId) || editor(farmId) || admin();
      }
    }
  }
}
